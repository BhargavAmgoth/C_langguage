///// Matrix multiplication
#include <stdio.h>

int main() {
    int r1,c1,r2,c2;
    
    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1,&c1);
    printf("Enter rows and columns of secound matrix: ");
    scanf("%d %d", &r2,&c2);
    
    if (c1 != r2) {
        printf("Matrix multiplication not possible.\n");
        return 0;
    }
    
    int a[r1][c1], b[r2][c2], res[r1][c2];
    
    printf("Enter elements of second matrix:\n");
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c1; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }
    
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c1; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }
    
 
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c2; j++)
        {
            res[i][j] = 0;
            for(int k=0; k<c1; k++)
            {
                res[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    printf("Resultant matrix:\n");
    for (int i = 0; i < r1; i++) 
    {
        for (int j = 0; j < c2; j++)
        {
             printf("%d ", res[i][j]);
        }
        printf("\n");
    }

    return 0;
}

//Reverse the word in a string


#include <stdio.h>
#include <string.h>

void reverse_word(char str[], int start, int end);

void reverse_word(char str[], int start, int end)
{
    while(start < end)
    {
        int temp = str[start];
        str[start] = str[end];
        str[end] = temp;
        start++;
        end--;
        
    }
}


int main() {
    
char str[] = "Try Code Execution in compiler";
    
    int str_len = strlen(str);
    printf("The str len is %d \n", str_len);
    
    int start = 0; 
    int i = 0;
    while(str[i] != '\0')
    {
        if(str[i] == ' ') {
            reverse_word(str, start, i-1);
            start = i+1;
        }
       i++;
    }
    reverse_word(str, start, i-1);
    printf("Reversed string: %s\n", str);
    return 0;
}
