// Find large number and secound large in an array
#include <stdio.h>

#define ARR_SIZE(X)   (sizeof(X) / sizeof(X[0]))


int main() {
   int arr[] = {2,4,5,7,8,5,3,6};
   
   int large = arr[0];
   int s_large = arr[0];
   for(int i=0; i<ARR_SIZE(arr); i++)
   {
      if(large < arr[i])
      {
          s_large = large;
          large = arr[i];
      }
   }
   
    printf("The large num is %d and the secound large is %d", large, s_large);
    return 0;
}

// Matrix multiplication
#include <stdio.h>

int main() {
    int r1,c1,r2,c2;
    
    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1,&c1);
    printf("Enter rows and columns of secound matrix: ");
    scanf("%d %d", &r2,&c2);
    
    if (c1 != r2) {
        printf("Matrix multiplication not possible.\n");
        return 0;
    }
    
    int a[r1][c1], b[r2][c2], res[r1][c2];
    
    printf("Enter elements of second matrix:\n");
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c1; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }
    
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c1; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }
    
 
    for(int i=0; i<r1; i++)
    {
        for(int j=0; j<c2; j++)
        {
            res[i][j] = 0;
            for(int k=0; k<c1; k++)
            {
                res[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    printf("Resultant matrix:\n");
    for (int i = 0; i < r1; i++) 
    {
        for (int j = 0; j < c2; j++)
        {
             printf("%d ", res[i][j]);
        }
        printf("\n");
    }

    return 0;
}
