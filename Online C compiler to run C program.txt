#include <stdio.h>

int main() {
    int total_sum = sum_num();
    
    printf("The sum of the first 100 natural numbers is: %d\n", total_sum);
    
    return 0;
}

int sum_num()
{
    static int n = 100;
    if(n == 0)
    {
         return 0; 
    }
    int current_n = n;
    n--;
    return current_n + sum_num(); 
}
